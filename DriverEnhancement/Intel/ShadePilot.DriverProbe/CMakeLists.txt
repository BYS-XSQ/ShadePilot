cmake_minimum_required(VERSION 3.13.4 FATAL_ERROR)
project(ShadePilot.DriverProbe VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(IGCL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../igcl-upstream")
add_executable(ShadePilot.DriverProbe main.cpp "${IGCL_ROOT}/Source/cApiWrapper.cpp")
add_executable(ShadePilot.DriverControl driver_control.cpp ../igcl-upstream/Source/cApiWrapper.cpp)
add_executable(ShadePilot.ScalingValidation scaling_validation.cpp ../igcl-upstream/Source/cApiWrapper.cpp)
add_executable(ShadePilot.CcdProbe ccd_probe.cpp)
add_executable(ShadePilot.DriverLab ../../tools/driver-probe/intel/src/intel_driverlab.cpp ../igcl-upstream/Source/cApiWrapper.cpp)
target_include_directories(ShadePilot.DriverProbe PRIVATE "${IGCL_ROOT}/include")
target_include_directories(ShadePilot.DriverControl PRIVATE ../igcl-upstream/include)
target_include_directories(ShadePilot.ScalingValidation PRIVATE ../igcl-upstream/include)
target_include_directories(ShadePilot.DriverLab PRIVATE ../igcl-upstream/include)
if(MSVC)
  target_compile_definitions(ShadePilot.DriverProbe PRIVATE UNICODE _UNICODE)
  target_compile_definitions(ShadePilot.DriverControl PRIVATE UNICODE _UNICODE)
  target_compile_definitions(ShadePilot.ScalingValidation PRIVATE UNICODE _UNICODE)
  target_compile_definitions(ShadePilot.CcdProbe PRIVATE UNICODE _UNICODE)
  target_compile_definitions(ShadePilot.DriverLab PRIVATE UNICODE _UNICODE)
  target_compile_options(ShadePilot.DriverLab PRIVATE /utf-8)
endif()
target_link_libraries(ShadePilot.CcdProbe PRIVATE user32)
